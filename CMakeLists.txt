cmake_minimum_required(VERSION 3.21)
project(nexus-media-player VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick Qml OpenGL)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv)
add_library(nexusmpvplugin MODULE src/plugin.cpp src/mpvitem.cpp src/theme.cpp)
target_link_libraries(nexusmpvplugin PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Qml Qt6::OpenGL PkgConfig::MPV)
set_target_properties(nexusmpvplugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qml/Nexus/Mpv")
configure_file(qml/qmldir.in "${CMAKE_BINARY_DIR}/qml/Nexus/Mpv/qmldir" COPYONLY)
